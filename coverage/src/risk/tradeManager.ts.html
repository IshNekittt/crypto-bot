/**

 * Long: entry + (tp - entry) * progress. Valid for shorts too, because tp < entry

function priceAtProgress(entry: Decimal, tp: Decimal, progress: Decimal): Decimal {
  return entry.plus(tp.minus(entry).mul(progress));
